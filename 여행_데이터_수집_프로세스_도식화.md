## 개요

Triplan의 RAG 시스템은 **실제 여행 경험 데이터**를 수집하여
사용자에게 검증된 여행 코스를 추천합니다.

| **YouTube 영상** | 10,000개 | 0개 | 100개/일 수집 |
| **Naver 블로그** | 3,000개 | 0개 | 50개/일 수집 |

```python
# Rate Limit
```

#### 수집 코드

```python
import requests
from datetime import datetime, timedelta

def collect_youtube_vlogs(
    query: str = "여행 vlog",
    max_results: int = 100,
    published_after: str = None
) -> list:
    """
    YouTube 여행 Vlog 수집
    """

    if published_after is None:
        # 최근 1년 이내 영상만
        published_after = (datetime.now() - timedelta(days=365)).isoformat() + "Z"

    results = []
    next_page_token = None

    while len(results) < max_results:
        # Search API 호출
        params = {
            "part": "snippet",
            "q": query,
            "type": "video",
            "maxResults": min(50, max_results - len(results)),
            "order": "viewCount",  # 조회수 높은 순
            "publishedAfter": published_after,
            "regionCode": "KR",  # 한국
            "relevanceLanguage": "ko",  # 한국어
            "key": API_KEY
        }

        if next_page_token:
            params["pageToken"] = next_page_token

        response = requests.get(f"{BASE_URL}/search", params=params)
        data = response.json()

        # 결과 파싱
        for item in data.get("items", []):
            video_id = item["id"]["videoId"]
            snippet = item["snippet"]

            # Video Statistics API 호출 (조회수, 좋아요)
            stats = get_video_statistics(video_id)

            results.append({
                "video_id": video_id,
                "title": snippet["title"],
                "description": snippet["description"],
                "channel": snippet["channelTitle"],
                "published_at": snippet["publishedAt"],
                "thumbnail": snippet["thumbnails"]["high"]["url"],
                "view_count": stats.get("viewCount", 0),
                "like_count": stats.get("likeCount", 0),
                "comment_count": stats.get("commentCount", 0),
            })

        # 다음 페이지 토큰
        next_page_token = data.get("nextPageToken")
        if not next_page_token:
            break

    return results


def get_video_statistics(video_id: str) -> dict:
    """
    YouTube 영상 통계 정보 가져오기
    """
    params = {
        "part": "statistics",
        "id": video_id,
        "key": API_KEY
    }

    response = requests.get(f"{BASE_URL}/videos", params=params)
    data = response.json()

    if data["items"]:
        return data["items"][0]["statistics"]
    return {}
```

#### 수집 코드

```python
import requests
from bs4 import BeautifulSoup

def collect_naver_blogs(
    query: str = "여행 후기",
    display: int = 100,
    sort: str = "sim"  # sim(유사도순), date(최신순)
) -> list:
    """
    Naver 블로그 여행 후기 수집
    """

    headers = {
        "X-Naver-Client-Id": CLIENT_ID,
        "X-Naver-Client-Secret": CLIENT_SECRET
    }

    params = {
        "query": query,
        "display": display,
        "sort": sort
    }

    response = requests.get(NAVER_API_URL, headers=headers, params=params)
    data = response.json()

    results = []

    for item in data.get("items", []):
        # HTML 태그 제거
        title = remove_html_tags(item["title"])
        description = remove_html_tags(item["description"])

        # 블로그 상세 페이지 크롤링
        blog_content = crawl_blog_content(item["link"])

        results.append({
            "post_id": extract_post_id(item["link"]),
            "title": title,
            "description": description,
            "content": blog_content,
            "author": item["bloggername"],
            "url": item["link"],
            "published_date": item["postdate"],
        })

    return results


def crawl_blog_content(url: str) -> str:
    """
    블로그 본문 크롤링
    """
    try:
        response = requests.get(url, timeout=10)
        soup = BeautifulSoup(response.text, "html.parser")

        # Naver 블로그 본문 선택자
        content = soup.select_one("#postViewArea") or soup.select_one(".se-main-container")

        if content:
            # 스크립트, 스타일 제거
            for script in content(["script", "style"]):
                script.decompose()

            return content.get_text(strip=True)

        return ""

    except Exception as e:
        print(f"Failed to crawl {url}: {e}")
        return ""


def remove_html_tags(text: str) -> str:
    """
    HTML 태그 제거
    """
    return BeautifulSoup(text, "html.parser").get_text()
```

### 3️⃣ Tistory 웹 크롤링

#### 크롤링 전략

```python
import requests
from bs4 import BeautifulSoup
import time

def collect_tistory_blogs(
    query: str = "여행 일정",
    max_pages: int = 10
) -> list:
    """
    Tistory 블로그 크롤링
    """

    results = []

    # Google 검색 활용 (site:tistory.com)
    search_url = f"https://www.google.com/search?q=site:tistory.com+{query}"

    # User-Agent 설정 (봇 차단 방지)
    headers = {
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"
    }

    for page in range(max_pages):
        params = {"start": page * 10}
        response = requests.get(search_url, headers=headers, params=params)
        soup = BeautifulSoup(response.text, "html.parser")

        # 검색 결과 링크 추출
        links = soup.select("a[href*='tistory.com']")

        for link in links[:10]:
            url = link.get("href")

            if url and "tistory.com" in url:
                # 블로그 본문 크롤링
                blog_data = crawl_tistory_post(url)

                if blog_data:
                    results.append(blog_data)

                # Rate Limit (1초 간격)
                time.sleep(1)

    return results


def crawl_tistory_post(url: str) -> dict:
    """
    Tistory 포스트 크롤링
    """
    try:
        response = requests.get(url, timeout=10)
        soup = BeautifulSoup(response.text, "html.parser")

        # 제목
        title = soup.select_one("h1") or soup.select_one(".title")
        title_text = title.get_text(strip=True) if title else ""

        # 본문
        content = soup.select_one(".article") or soup.select_one(".entry-content")
        content_text = content.get_text(strip=True) if content else ""

        # 작성일
        date = soup.select_one(".date") or soup.select_one("time")
        date_text = date.get_text(strip=True) if date else ""

        # 이미지 수
        images = soup.select("img")
        image_count = len(images)

        # 필터링: 최소 조건 확인
        if len(content_text) < 1000 or image_count < 5:
            return None

        return {
            "post_id": extract_tistory_id(url),
            "title": title_text,
            "content": content_text,
            "url": url,
            "published_date": date_text,
            "image_count": image_count
        }

    except Exception as e:
        print(f"Failed to crawl {url}: {e}")
        return None
```

```python
import re
from bs4 import BeautifulSoup

def clean_text(text: str) -> str:
    """
    HTML 태그 및 특수문자 제거
    """
    # 1. HTML 태그 제거
    text = BeautifulSoup(text, "html.parser").get_text()

    # 2. HTML 엔티티 디코딩
    text = text.replace("&nbsp;", " ")
    text = text.replace("&quot;", '"')
    text = text.replace("&amp;", "&")

    # 3. 이모지 제거 (선택적)
    # emoji_pattern = re.compile("["
    #     u"\U0001F600-\U0001F64F"  # 이모티콘
    #     u"\U0001F300-\U0001F5FF"  # 기호
    #     "]+", flags=re.UNICODE)
    # text = emoji_pattern.sub(r'', text)

    # 4. 과도한 공백 제거
    text = re.sub(r'\s+', ' ', text)

    # 5. 줄바꿈 정규화
    text = re.sub(r'\n+', '\n', text)

    return text.strip()
```

```python
import re
from typing import List, Dict

def parse_schedule_advanced(text: str) -> List[Dict]:
    """
    여행 일정 파싱 (고급 버전)
    """

    schedules = []

    # 패턴 1: "Day 1:", "Day 2:" 등
    day_pattern = r'Day\s*(\d+)\s*[:\-](.+?)(?=Day\s*\d+|$)'
    matches = re.finditer(day_pattern, text, re.IGNORECASE | re.DOTALL)

    for match in matches:
        day_no = int(match.group(1))
        day_content = match.group(2).strip()

        schedule = parse_day_content(day_no, day_content)
        schedules.append(schedule)

    # 패턴 2: "첫째 날:", "둘째 날:" 등 (한글)
    if not schedules:
        korean_pattern = r'(첫째|둘째|셋째|넷째|다섯째)\s*날\s*[:\-](.+?)(?=(첫째|둘째|셋째|넷째|다섯째)\s*날|$)'
        korean_days = {"첫째": 1, "둘째": 2, "셋째": 3, "넷째": 4, "다섯째": 5}

        matches = re.finditer(korean_pattern, text, re.DOTALL)

        for match in matches:
            day_korean = match.group(1)
            day_no = korean_days[day_korean]
            day_content = match.group(2).strip()

            schedule = parse_day_content(day_no, day_content)
            schedules.append(schedule)

    return sorted(schedules, key=lambda x: x['day'])


def parse_day_content(day_no: int, content: str) -> Dict:
    """
    하루 일정 내용 파싱
    """

    # 장소 추출 (쉼표, 화살표, 줄바꿈으로 구분)
    places = re.split(r'[,\→\→\n]', content)
    places = [p.strip() for p in places if p.strip()]
    places = [p for p in places if 2 < len(p) < 50]

    # 시간 추출 (HH:MM 패턴)
    time_pattern = r'(\d{1,2}:\d{2})'
    times = re.findall(time_pattern, content)

    # 활동 추출 (동사 패턴)
    activity_keywords = ["감상", "촬영", "투어", "방문", "탐험", "걷기", "이동"]
    activities = []
    for keyword in activity_keywords:
        if keyword in content:
            # 주변 문맥 추출
            pattern = rf'([^.!?\n]{{10,50}}{keyword}[^.!?\n]{{0,20}})'
            matches = re.findall(pattern, content)
            activities.extend([m.strip() for m in matches])

    # 예산 추출 (금액 패턴)
    budget_pattern = r'(\d{1,3}(?:,\d{3})*)\s*원'
    budget_matches = re.findall(budget_pattern, content)
    budget_total = sum(int(b.replace(',', '')) for b in budget_matches)

    return {
        "day": day_no,
        "places": places[:10],  # 최대 10개
        "times": times[:10],
        "activities": activities[:10],
        "budget_krw": budget_total if budget_total > 0 else None
    }
```

```python
def validate_data(data: dict) -> tuple[bool, list]:
    """
    데이터 유효성 검사

    Returns:
        (is_valid, errors)
    """
    errors = []

    # 1. 필수 필드 확인
    required_fields = ["video_id", "title", "embedding_text"]
    for field in required_fields:
        if not data.get(field):
            errors.append(f"Missing required field: {field}")

    # 2. 벡터 차원 확인
    if "content_embedding" in data:
        if len(data["content_embedding"]) != 1536:
            errors.append(f"Invalid embedding dimension: {len(data['content_embedding'])}")

    # 3. 지역 코드 유효성
    if data.get("country_code"):
        if data["country_code"] not in VALID_COUNTRY_CODES:
            errors.append(f"Invalid country code: {data['country_code']}")

    # 4. 날짜 범위 확인
    if data.get("upload_year"):
        current_year = datetime.now().year
        if data["upload_year"] < 2020 or data["upload_year"] > current_year:
            errors.append(f"Invalid year: {data['upload_year']}")

    # 5. 조회수 범위
    if data.get("views_num"):
        if data["views_num"] < 0 or data["views_num"] > 100000000:
            errors.append(f"Suspicious view count: {data['views_num']}")

    # 6. 제목 길이
    if data.get("title"):
        if len(data["title"]) < 10 or len(data["title"]) > 200:
            errors.append(f"Invalid title length: {len(data['title'])}")

    # 7. 메타데이터 구조
    if data.get("metadata"):
        if not isinstance(data["metadata"], dict):
            errors.append("Metadata must be a dictionary")

        # 일정 존재 확인
        schedules = data["metadata"].get("schedules", [])
        if not schedules or len(schedules) == 0:
            errors.append("No schedules found in metadata")

    return (len(errors) == 0, errors)
```

```python
from apps.ai.models import TripCourseEmbedding
from pgvector.django import CosineDistance

def remove_duplicates(new_data: dict) -> bool:
    """
    중복 데이터 제거

    Returns:
        True if duplicate, False otherwise
    """

    # 1. video_id 중복 체크 (DB)
    if TripCourseEmbedding.objects.filter(video_id=new_data["video_id"]).exists():
        print(f"Duplicate video_id: {new_data['video_id']}")
        return True

    # 2. 벡터 유사도 중복 체크 (99% 이상 유사)
    if "content_embedding" in new_data:
        similar = TripCourseEmbedding.objects.annotate(
            distance=CosineDistance('content_embedding', new_data["content_embedding"])
        ).filter(distance__lt=0.01).first()  # distance < 0.01 = similarity > 0.99

        if similar:
            print(f"Duplicate by similarity: {similar.video_id} (distance: {similar.distance})")
            return True

    return False
```

```python
from django.db import transaction
from apps.ai.models import TripCourseEmbedding

@transaction.atomic
def bulk_save_embeddings(data_list: list, batch_size: int = 100):
    """
    배치로 DB 저장 (트랜잭션 처리)
    """

    objects = []

    for data in data_list:
        # 검증
        is_valid, errors = validate_data(data)
        if not is_valid:
            print(f"Validation failed for {data.get('video_id')}: {errors}")
            continue

        # 중복 체크
        if remove_duplicates(data):
            continue

        # 객체 생성
        obj = TripCourseEmbedding(
            video_id=data["video_id"],
            title=data["title"],
            channel=data.get("channel"),
            url=data.get("url"),
            country_name=data.get("country_name"),
            country_code_id=data.get("country_code"),
            region1_idx_id=data.get("region1_idx"),
            region2_idx_id=data.get("region2_idx"),
            upload_year=data.get("upload_year"),
            upload_month=data.get("upload_month"),
            views_num=data.get("views_num"),
            content_embedding=data.get("content_embedding"),
            metadata=data.get("metadata")
        )
        objects.append(obj)

        # 배치 크기 도달 시 저장
        if len(objects) >= batch_size:
            TripCourseEmbedding.objects.bulk_create(objects, ignore_conflicts=True)
            print(f"Saved {len(objects)} objects")
            objects = []

    # 남은 객체 저장
    if objects:
        TripCourseEmbedding.objects.bulk_create(objects, ignore_conflicts=True)
        print(f"Saved {len(objects)} objects")
```
